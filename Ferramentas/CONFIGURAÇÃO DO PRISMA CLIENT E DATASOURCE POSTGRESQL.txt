// ==============================================================================
// CONFIGURAÇÃO DO PRISMA CLIENT E DATASOURCE POSTGRESQL
// ==============================================================================

generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

// ==============================================================================
// 1. GESTÃO DE UNIDADES E ALIASES (SINÔNIMOS)
// ==============================================================================

model Unidade {
  id             String   @id @default(uuid())
  nome           String   @unique // Nome de exibição principal (Ex: "Santana de Parnaíba")
  nomeSanitizado String   @unique // Nome sem acentos e em maiúsculas (Ex: "SANTANA DE PARNAIBA")
  createdAt      DateTime @default(now())
  updatedAt      DateTime @updatedAt

  // Relacionamentos
  aliases          UnidadeAlias[]
  impressoras      Impressora[]
  leituras         LeituraImpressora[]
  perfilFranquia   PerfilFranquia?     @relation(fields: [perfilFranquiaId], references: [id], onDelete: SetNull)
  perfilFranquiaId String?

  @@index([nomeSanitizado])
}

model UnidadeAlias {
  id        String   @id @default(uuid())
  alias     String   @unique // String sanitizada secundária (Ex: "SANTANA", "PARNAIBA")
  unidadeId String
  unidade   Unidade  @relation(fields: [unidadeId], references: [id], onDelete: Cascade)
  createdAt DateTime @default(now())

  @@index([alias])
}

// ==============================================================================
// 2. PARQUE DE IMPRESSORAS (EQUIPAMENTOS)
// ==============================================================================

model Impressora {
  id        String   @id @default(uuid())
  serial    String   @unique // Serial único retornado via SNMP
  modelo    String
  ip        String
  apelido   String?  // Apelido/Setor editável pelo usuário (Ex: "KONICA RECEPÇÃO")
  unidadeId String
  unidade   Unidade  @relation(fields: [unidadeId], references: [id], onDelete: Cascade)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt

  leituras LeituraImpressora[]

  @@index([unidadeId])
  @@index([ip])
}

// ==============================================================================
// 3. REGISTRO DE LEITURAS (INGESTÃO DOS CSVS E CÁLCULO DE DELTA)
// ==============================================================================

model LeituraImpressora {
  id           String     @id @default(uuid())
  dataHora     DateTime   // Carimbo de data e hora lido do arquivo CSV
  paginasTotal Int
  paginasPB    Int
  paginasColor Int
  status       String     // "Online", "Offline", "Erro SNMP"
  resetPlaca   Boolean    @default(false) // Flag de segurança ativada se C_final < C_inicial

  // Chaves de identificação
  ip           String
  serial       String
  impressoraId String?
  impressora   Impressora? @relation(fields: [impressoraId], references: [id], onDelete: SetNull)
  unidadeId    String
  unidade      Unidade     @relation(fields: [unidadeId], references: [id], onDelete: Cascade)

  createdAt    DateTime   @default(now())

  // Bloqueio de leituras idênticas no mesmo instante
  @@unique([ip, serial, dataHora])
  
  // Índices de alta performance para o cálculo diferencial por Delta no BI
  @@index([unidadeId, dataHora])
  @@index([impressoraId, dataHora])
  @@index([serial, dataHora])
}

// ==============================================================================
// 4. MÓDULO FINANCEIRO (CONTRATOS E FRANQUIAS)
// ==============================================================================

model PerfilFranquia {
  id                String    @id @default(uuid())
  nome              String    @unique // Ex: "Contrato Padrão Filiais", "Contrato Especial Matriz"
  tarifaAvulsaPB    Float     @default(0.04)
  tarifaAvulsaColor Float     @default(0.25)

  // Regras de Franquia Fixa Mensal
  temFranquiaFixa   Boolean   @default(true)
  valorFixoMensal   Float?    @default(500.0)
  cotaPB            Int?      @default(5000)
  cotaColor         Int?      @default(1000)
  excedentePB       Float?    @default(0.05)
  excedenteColor    Float?    @default(0.30)

  // Insumos
  precoResmaPapel   Float     @default(28.0)

  unidades          Unidade[]
  createdAt         DateTime  @default(now())
  updatedAt         DateTime  @updatedAt
}

// ==============================================================================
// 5. AUDITORIA DE SLA DE ENVIO E ALERTAS
// ==============================================================================

model ConfiguracaoSLA {
  id                  String   @id @default("default_config")
  horarioLimite       String   @default("17:00") // Formato HH:mm
  alertasAtivos       Boolean  @default(true)
  emailsDestinatarios String   @default("") // E-mails separados por vírgula
  updatedAt           DateTime @updatedAt
}

// ==============================================================================
// 6. CONTROLADOR DE ARQUIVOS PROCESSADOS DO GOOGLE DRIVE
// ==============================================================================

model ArquivoIngerido {
  id                String   @id @default(uuid())
  nomeArquivo       String   @unique // Ex: "relatorio_impressoras_Santana_2026-08-01_17-00-00.csv"
  dataProcessamento DateTime @default(now())
  tamanhoBytes      BigInt
  status            String   @default("PROCESSADO") // "PROCESSADO", "ERRO"
  erroMensagem      String?

  @@index([nomeArquivo])
}
